def labelled(box, rr, path, alt, caption=None, labels=()):
    """A picture with arrows to its parts: labels = [(fx, fy, text, lx, ly)] as fractions of the picture."""
    img = box.picture(rr, path, alt, caption=caption)
    for fx, fy, text, lx, ly in labels:
        box.annotate(img, fx, fy, text, lx, ly)
    return img


def mind_map(page, r, centre, branches, size=None):
    """A mind map drawn with the design system's shapes: a centre node, branches = [(title, [points])] placed around it.
    size: text size of the points (default small + 1)."""
    f = page.fmt
    size = size or f.small + 1
    G = f.G
    spots = [(0.0, 0.0), (0.0, 0.53), (0.68, 0.0), (0.68, 0.53), (0.34, 0.6)]
    w = r.w * 0.32
    cbox = Rect(r.x + r.w * 0.36, r.y + r.h * 0.14, r.w * 0.28, r.h * 0.3)
    nodes = []
    for (title, points), (sx, sy), colour in zip(branches, spots, BOX_COLOURS + (BOX_COLOURS[0],)):
        h = r.h * (0.47 if sx != 0.34 else 0.4)
        nodes.append((Rect(r.x + sx * r.w, r.y + sy * r.h, w, h), title, points, colour))
    for node, _t, _p, colour in nodes:               # lines first, so the boxes sit on top
        ln = page.slide.shapes.add_connector(MSO_CONNECTOR.STRAIGHT, Pt(cbox.x + cbox.w / 2), Pt(cbox.y + cbox.h / 2),
                                             Pt(node.x + node.w / 2), Pt(node.y + node.h / 2))
        ln.line.color.rgb = __import__("pptx.dml.color", fromlist=["RGBColor"]).RGBColor.from_string(colour)
        ln.line.width = Pt(2)
    _shape(page.slide, __import__("pptx.enum.shapes", fromlist=["MSO_SHAPE"]).MSO_SHAPE.ROUNDED_RECTANGLE, cbox, MAIN,
           radius=G)
    _text(page.slide, cbox.inset(0.6 * G), [centre], f.sub + 3, color=WHITE, bold=True, anchor=MSO_ANCHOR.MIDDLE,
          align=PP_ALIGN.CENTER, after=0)
    for node, title, points, colour in nodes:
        _shape(page.slide, __import__("pptx.enum.shapes", fromlist=["MSO_SHAPE"]).MSO_SHAPE.ROUNDED_RECTANGLE, node,
               ACCENT[colour], colour, 1.5, radius=0.6 * G)
        _text(page.slide, node.inset(0.5 * G, 0.4 * G), [{"text": title, "size": f.sub, "color": colour, "bold": True,
                                                          "after": 2}] + ["- " + x for x in points], size, after=1,
              bullet_color=colour)


def dashed(page, r, label):
    s = _shape(page.slide, MSO_SHAPE.ROUNDED_RECTANGLE, r, WHITE, MAIN, 1.5, radius=page.fmt.G)
    s.line.dash_style = 4
    _text(page.slide, r.inset(page.fmt.G), [label], page.fmt.body, anchor=MSO_ANCHOR.MIDDLE, align=PP_ALIGN.CENTER)

def split_last_cell(r, page, cols, rows):
    """The empty last cell of a flow grid (same split as Box.flow), for a note beside the second row."""
    from design_system import split
    f = page.fmt
    row = split(r, [1] * rows, 0.6 * f.G, vertical=True)[-1]
    cell = split(row, [1] * cols, 1.6 * f.G)[-1]
    return cell.inset(0, 0.1 * cell.h, 0, 0.25 * cell.h)
